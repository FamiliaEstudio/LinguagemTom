cmake_minimum_required(VERSION 3.21)
project(tom_runtime VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL3 CONFIG REQUIRED)
find_package(SDL3_ttf CONFIG REQUIRED)
find_path(MPDEC_INCLUDE_DIR mpdecimal.h REQUIRED)
find_library(MPDEC_LIBRARY NAMES mpdec REQUIRED)
find_path(YYJSON_INCLUDE_DIR yyjson.h REQUIRED)
find_library(YYJSON_LIBRARY NAMES yyjson REQUIRED)
find_path(SQLITE_INCLUDE_DIR sqlite3.h REQUIRED)
find_library(SQLITE_LIBRARY NAMES sqlite3 REQUIRED)
find_path(UTF8PROC_INCLUDE_DIR utf8proc.h REQUIRED)
find_library(UTF8PROC_LIBRARY NAMES utf8proc REQUIRED)
find_path(MINIZ_INCLUDE_DIR miniz.h PATH_SUFFIXES miniz REQUIRED)
find_library(MINIZ_LIBRARY NAMES miniz REQUIRED)
find_path(LIBXML_INCLUDE_DIR libxml/parser.h PATH_SUFFIXES libxml2 REQUIRED)
find_library(LIBXML_LIBRARY NAMES xml2 libxml2s REQUIRED)
add_library(tom_text STATIC common.c text.c)
add_library(tom_decimal STATIC decimal.c)
add_library(tom_ui STATIC ui.c)
add_library(tom_platform STATIC platform.c)
add_library(tom_math STATIC math.c)
add_library(tom_audio STATIC audio.c)
add_library(tom_data STATIC data.c)
add_library(tom_json STATIC json.c)
add_library(tom_channel STATIC channel.c)
add_library(tom_sqlite STATIC sqlite.c)
add_library(tom_document STATIC document.c)
add_library(tom_editor STATIC editor.c)
add_library(tom_editor_sqlite STATIC editor_sqlite.c)
add_library(tom_files STATIC files.c)
add_library(tom_docx STATIC docx.c)
add_library(tom_file_jobs STATIC file_jobs.c)
add_library(tom_file_dialog STATIC file_dialog.c)
add_library(tom_form STATIC form.c)
foreach(target tom_text tom_decimal tom_ui tom_platform tom_math tom_audio tom_data tom_json tom_channel tom_sqlite tom_document tom_editor tom_editor_sqlite tom_files tom_docx tom_file_jobs tom_file_dialog tom_form)
  target_include_directories(${target} PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>" "$<INSTALL_INTERFACE:include>")
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
endforeach()
target_include_directories(tom_decimal PRIVATE "${MPDEC_INCLUDE_DIR}")
target_link_libraries(tom_decimal PUBLIC tom_text "${MPDEC_LIBRARY}")
target_link_libraries(tom_platform PUBLIC tom_text SDL3::SDL3)
target_link_libraries(tom_ui PUBLIC tom_platform SDL3_ttf::SDL3_ttf)
target_link_libraries(tom_audio PUBLIC tom_platform)
target_link_libraries(tom_math PUBLIC tom_text)
target_link_libraries(tom_data PUBLIC tom_platform)
target_link_libraries(tom_channel PUBLIC tom_platform)
target_include_directories(tom_json PRIVATE "${YYJSON_INCLUDE_DIR}")
target_link_libraries(tom_json PUBLIC tom_decimal "${YYJSON_LIBRARY}")
target_include_directories(tom_sqlite PRIVATE "${SQLITE_INCLUDE_DIR}")
target_link_libraries(tom_sqlite PUBLIC tom_text "${SQLITE_LIBRARY}" "${UTF8PROC_LIBRARY}")
target_include_directories(tom_sqlite PRIVATE "${UTF8PROC_INCLUDE_DIR}")
target_compile_definitions(tom_sqlite PRIVATE UTF8PROC_STATIC)
target_include_directories(tom_document PRIVATE "${UTF8PROC_INCLUDE_DIR}" "${YYJSON_INCLUDE_DIR}")
target_compile_definitions(tom_document PRIVATE UTF8PROC_STATIC)
target_link_libraries(tom_document PUBLIC tom_text "${UTF8PROC_LIBRARY}" "${YYJSON_LIBRARY}")
target_include_directories(tom_editor PRIVATE "${UTF8PROC_INCLUDE_DIR}")
target_compile_definitions(tom_editor PRIVATE UTF8PROC_STATIC)
target_link_libraries(tom_editor PUBLIC tom_document tom_ui)
target_include_directories(tom_editor_sqlite PRIVATE "${SQLITE_INCLUDE_DIR}" "${YYJSON_INCLUDE_DIR}")
target_link_libraries(tom_files PUBLIC tom_text tom_platform)
target_include_directories(tom_docx PRIVATE "${MINIZ_INCLUDE_DIR}" "${LIBXML_INCLUDE_DIR}" "${UTF8PROC_INCLUDE_DIR}")
target_compile_definitions(tom_docx PRIVATE MINIZ_STATIC_DEFINE LIBXML_STATIC UTF8PROC_STATIC)
target_link_libraries(tom_docx PUBLIC tom_document tom_files "${MINIZ_LIBRARY}" "${LIBXML_LIBRARY}")
target_include_directories(tom_file_jobs PRIVATE "${YYJSON_INCLUDE_DIR}" "${SQLITE_INCLUDE_DIR}")
target_link_libraries(tom_file_jobs PUBLIC tom_docx tom_sqlite)
target_include_directories(tom_file_dialog PRIVATE "${YYJSON_INCLUDE_DIR}")
target_link_libraries(tom_file_dialog PUBLIC tom_platform "${YYJSON_LIBRARY}")
target_include_directories(tom_form PRIVATE "${YYJSON_INCLUDE_DIR}")
target_link_libraries(tom_form PUBLIC tom_editor)
target_link_libraries(tom_editor_sqlite PUBLIC tom_document tom_sqlite tom_platform)
if(WIN32)
  target_link_libraries(tom_docx PUBLIC bcrypt)
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(tom_sqlite PUBLIC Threads::Threads m)
  target_link_libraries(tom_ui PUBLIC m)
  target_link_libraries(tom_editor PUBLIC m)
  target_link_libraries(tom_math PUBLIC m)
  target_link_libraries(tom_audio PUBLIC m)
endif()
install(TARGETS tom_text tom_decimal tom_ui tom_platform tom_math tom_audio tom_data tom_json tom_channel tom_sqlite tom_document tom_editor tom_editor_sqlite tom_files tom_docx tom_file_jobs tom_file_dialog tom_form ARCHIVE DESTINATION lib)
install(FILES tom_runtime.h DESTINATION include)
